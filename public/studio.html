اللايفات الجاية

          
ما كاين حتى لايف مبرمج.

          
    
          
        
    
        
    
          
    الرابط ديال القاعة
    
          
    هاد الرابط كتعطيه للناس باش يتفرجو.
    
          
    
          
    
        
    
      
    
    
    
    
    
      
      
      
      
      الإدارة
      
    
    
    
    
      
      
      
    الكاميرا
    
      
      
    
      
    على الهواء
    
    
    
    
    
    
    
      
    
        
        
        
        
      
    
      
    
        
        
        
        
      
    
      
      
      
      
      
        
        
        
        
        
        
        
      
      
      
    
    
    
    جاهز. اختار المصدر وبدا اللايف.
    
    
    

    
    
      
      
      
      
      
      
      
      
    
    

    
    
      
    الأسئلة
    
      
    الأسئلة اللي كيصيفطو المتابعين بزر «سول الأستاذ» كيبانو هنا، بعاد على التعليقات.
    
      
      
    
      
    
      
      
      تصويت ولا كويز
      
      
      
        
        
        
      
        
      كتب جوج اختيارات ولا أكثر. إلا بغيتي كويز، علم على الجواب الصحيح.
      
        
      
          
          
          
        
      
      
      
      
      
      
      
    
      

    
      
      
      المنصة
      
      
      ما كاين حتى واحد على المنصة. تقدر تطلع حتى ل 12 واحد معاك.
      
      
        
      
      
        طلبات الطلوع
        
      
        ما كاين حتى طلب دابا.
        
      
          
    
          

    
          
      
          المساعدين
          
      
          المساعد كيدخل بالحساب ديالو كمتابع، وكيقدر يقبل ولا يرفض الناس، يطلع وينزل الضيوف، يسد الميكرو، يحذف التعليقات ويحظر. ما كيقدرش يبدا اللايف ولا يبدل الإعدادات.
          
      
          
        
        
        
      
          
      
          
      
            
    
            

    
            
      
            القسم ديالي
            
      
            الدخول: مجانا ولا بالخلاص
            
      
            
        
        
        
        
        
        
      
            
      
            
      
      
      
            التلاميذ
            
      
            كل واحد دخل للقاعة بالحساب ديالو كيتزاد هنا بوحدو. التلاميذ ما كيتطلبش منهم الكود مرة أخرى.
            
      
            
        
        
        
      
            
      
            
      
              
      
              وثائق للتحميل
              
      
              PDF، Word، PowerPoint، Excel، صور… التلاميذ كيلقاوهم فالقاعة وكيحملوهم. حتى ل 20 ملف، 30 ميغا لكل واحد.
              
      
              
      
              
      
                
      
                ورقة الحضور
                
      
                من بعد كل لايف، كتلقى هنا شكون حضر وشحال بقى.
                
      
                  
    
                  

    
                  
      
                  الترجمة الفورية
                  
      
                  الهضرة ديالك كتكتب وكتترجم للعربية والفرنسية والإنجليزية. كل متابع كيختار اللغة ديالو: يقراها تحت الفيديو، ولا يسمعها بالصوت.
                  
      
                  
        
        
        
      
                  
      
                  
      
                  
    
                  

    
                  
      
                  
        
                  الأجندة
                  
        
      
                  
      
                  ما كاين حتى لايف مبرمج. ورك على «برمج لايف» باش تزيد واحد.
                  
      
                    
      
                    الدخول بالكود
                    
      
      
                    
      
                    هاد الكود كيحل القاعة فأي وقت، وكيتبدل مور كل لايف. كل لايف مبرمج عندو حتى هو الكود ديالو والرابط ديالو، وكيخدمو غير فالوقت ديالو (من نص ساعة قبل حتى لنص ساعة بعد).
                    
      
      
    
                    

    
                    
      
                    قاعة الانتظار
                    
      
      
      
                    ما كاين حتى واحد كيتسنى.
                    
      
                      
      
    
                      

    
                      
      
                      التسجيلات
                      
      
                      ما كاين حتى تسجيل. اللايف كيتسجل غير إلا وركتي على «سجل اللايف».
                      
      
                        
    
                        

    
                        
      
                        الإعدادات
                        
      
                        
        
        
        
      
                        
      
                        
        
        
        
      
                        
      
                        
        
        
      
                        
      
      
                        
        
        
        
      
                        
      
                        الصوت اللي كيسمعوه الناس ملي ما كاينش لايف. بلا ملف ديالك كيتسمعو سور الرحمن ثم الواقعة ثم الحديد بصوت عبد الباسط عبد الصمد، وحدة مور وحدة. تقدر تبدلهم بملف MP3 أو M4A (حتى ل 60 ميغا).
                        
      
      
      
                        
        
        
      
                        
      
                        الصور اللي كيشوفوها الناس ملي ما كاينش لايف، وحدة مور وحدة. بلا صور ديالك كتبان صورة البحر. حتى ل 12 صورة (JPG، PNG، WebP).
                        
      
                          
      
                          خلفيات الموقع (4K)
                          
      
                          علم على الخلفيات اللي بغيتي يبانو ملي ما كاينش لايف. كيدوزو وحدة مور وحدة، ومن بعدهم الصور ديالك.

                            الخلفية مورايا
